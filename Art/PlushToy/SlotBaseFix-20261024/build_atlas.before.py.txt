def _thin_base(_id, cut_frac, keep=.6, gamma=.6):
    if (PLUSH_FULL / f'win/{_id}.png').exists(): return f'win/{_id}.png'
    # 12:13 (user: like the 重排 button): the base ends up ~8 % of the height (green keeps 60 % of its base, the square
    # buttons 90 %) and is lifted (gamma) so it reads as a slightly deeper tone of the face, not a dark slab.
    import numpy as np
    src = DEL / f'win/{_id}.png'
    if not src.exists(): return f'win/{_id}.png'
    im = Image.open(src).convert('RGBA'); w, h = im.size; cut = round(h * cut_frac)
    base = im.crop((0, cut, w, h)); nb = max(1, round(base.height * keep))
    out = Image.new('RGBA', (w, h))
    out.alpha_composite(im.crop((0, 0, w, cut)).resize((w, h - nb), Image.LANCZOS), (0, 0))
    # 12:05 (user: the squashed base read too dark): the base is lifted a little (gamma .75 on its colours), since the
    # squash also squeezed out its light reflection line.
    _b = np.asarray(base.resize((w, nb), Image.LANCZOS)).astype(np.float32) / 255; _b[..., :3] = _b[..., :3] ** gamma
    out.alpha_composite(Image.fromarray((_b * 255).astype(np.uint8), 'RGBA'), (0, h - nb))
    # 14:24 (user: the base was lifted but the dark rim along the top and sides was not): the same lift on the dark
    # outline pixels everywhere (weighted by darkness, so the face is untouched).
    _a = np.asarray(out).astype(np.float32) / 255; _v = _a[..., :3].max(-1)
    _w = np.clip((.62 - _v) / .35, 0, 1)[..., None]
    _a[..., :3] = _a[..., :3] * (1 - _w) + (_a[..., :3] ** gamma) * _w
    out = Image.fromarray((_a * 255).astype(np.uint8), 'RGBA')
    dst = DEL / f'../generated/win/{_id}.png'; dst.parent.mkdir(parents=True, exist_ok=True); out.save(dst)
    return f'../generated/win/{_id}.png'

    # The outline takes the darkest ranks, a shade under 重排's own rim ((0,60,215) vs (1,80,233)): lift the darkest

# 2026-10-17: the H13 bubble is fully clear inside (reads as a hollow ring); a faint pale-rainbow film and a soft
# sheen are laid under it so it reads as a soap bubble. Also applied to the first pop frame (the whole bubble).
import numpy as np
from PIL import ImageFilter as _IF
def add_film(img, strength=1.0):
    img = img.convert('RGBA'); w, h = img.size
    a = np.array(img.getchannel('A'))
    ys, xs = np.nonzero(a > 40)
    cx, cy = (xs.min() + xs.max()) / 2, (ys.min() + ys.max()) / 2
    R = min(xs.max() - xs.min(), ys.max() - ys.min()) / 2 * .96
    Y, X = np.mgrid[0:h, 0:w]
    dx, dy = X - cx, Y - cy
    r = np.sqrt(dx * dx + dy * dy) / R
    ang = (np.arctan2(dy, dx) / (2 * np.pi)) % 1
    hue = (ang + .18 * np.sin(r * np.pi * 2.2) + .1 * np.sin(ang * 6 * np.pi)) % 1
    rgb = np.zeros((h, w, 3))
    hh = hue * 6; i = np.floor(hh).astype(int) % 6; f = hh - np.floor(hh); s = .45; v = 1.0
    p = v * (1 - s); q = v * (1 - s * f); t = v * (1 - s * (1 - f))
    for k, (rr, gg, bb) in enumerate([(v, t, p), (q, v, p), (p, v, t), (p, q, v), (t, p, v), (v, p, q)]):
        m = i == k
        rgb[..., 0] = np.where(m, rr if np.isscalar(rr) else rr, rgb[..., 0])
        rgb[..., 1] = np.where(m, gg if np.isscalar(gg) else gg, rgb[..., 1])
        rgb[..., 2] = np.where(m, bb if np.isscalar(bb) else bb, rgb[..., 2])
    alpha = (.07 + .26 * np.clip(r, 0, 1) ** 3) * strength
    alpha[r > 1] = 0
    rgb = rgb * .55 + .45
    sheen = np.clip(1 - np.sqrt(((X - (cx - R * .35)) / (R * .75)) ** 2 + ((Y - (cy - R * .4)) / (R * .55)) ** 2), 0, 1) ** 1.5 * .22
    alpha = np.maximum(alpha, sheen * (r <= 1)); rgb = rgb * (1 - sheen[..., None]) + sheen[..., None]
    film = np.dstack([rgb * 255, alpha * 255]).clip(0, 255).astype('uint8')
    film = Image.fromarray(film).filter(_IF.GaussianBlur(w / 160))
    out = film.copy(); out.alpha_composite(img); return out

# H14 celebration halo behind showcased members (victory screen, codex); optional until delivered.
# 2026-10-17 (user: the light is right, the outer bubbles too big): the bubble-free core is kept and its rays are
# continued outward (clean_halo); the program circles smaller bubbles around it. One of the halo's big bubbles
# (milky iridescent inside, user preferred it to the H13 one) becomes the home/codex soap bubble and pop frame 0.
def clean_halo(img, r0=200.0, r1=430.0):
    """Keep the bubble-free core of the H14 halo (r < r0) and continue its rays outward: beyond r0 every pixel
    takes the colour of the core's edge at the same angle, fading smoothly to nothing at r1."""
    im = np.array(img.convert('RGBA')).astype(np.float32)
    h, w = im.shape[:2]
    Y, X = np.mgrid[0:h, 0:w]
    cx, cy = (w - 1) / 2, (h - 1) / 2
    dx, dy = X - cx, Y - cy
    R = np.sqrt(dx * dx + dy * dy) + 1e-6
    pm = im.copy(); pm[..., :3] *= pm[..., 3:4] / 255
    # sample the ring at r0 (bilinear) along each pixel's angle
    sx = cx + dx / R * r0; sy = cy + dy / R * r0
    x0 = np.floor(sx).astype(int); y0 = np.floor(sy).astype(int); fx = (sx - x0)[..., None]; fy = (sy - y0)[..., None]
    def g(yy, xx): return pm[np.clip(yy, 0, h - 1), np.clip(xx, 0, w - 1)]
    edge = (g(y0, x0) * (1 - fx) * (1 - fy) + g(y0, x0 + 1) * fx * (1 - fy) + g(y0 + 1, x0) * (1 - fx) * fy + g(y0 + 1, x0 + 1) * fx * fy)
    u = np.clip((R - r0) / (r1 - r0), 0, 1); fall = (1 - u) ** 2.2
    ext = edge * fall[..., None]
    t = np.clip((R - (r0 - 14)) / 14, 0, 1)[..., None]   # 14 px blend into the extension
    out = pm * (1 - t) + ext * t
    a = out[..., 3:4]
    rgb = np.where(a > 0, out[..., :3] / np.maximum(a, 1e-3) * 255, 0)
    return Image.fromarray(np.dstack([np.clip(rgb, 0, 255), np.clip(a, 0, 255)]).astype(np.uint8))

def calm_stage(im, keep=.25):
    """2026-10-21 23:33 (user): the side cloud frame of the lower stage is filled with the stage's own gradient (narrow
    phones cropped those clouds into half circles at the screen edges), and the cloud band right behind the guard rail
    is thinned to a faint trace (`keep`), so the rail is the one clear edge between the field and the stage.
    Rows/columns are those of the 1080x1920 delivery (scaled for other sizes); the bottom cloud row is kept."""
    a = np.asarray(im.convert('RGB')).astype(np.float32)
    H, W = a.shape[:2]; k = H / 1920
    y0, y1, y2 = int(1100 * k), int(1850 * k), int(1900 * k)
    ref = a[:, int(300 * k):int(780 * k)].mean(1)                     # per-row stage colour
    x = np.arange(W)[None, :]
    inner, outer = 125 * k, 70 * k
    side = np.clip((inner - np.minimum(x, W - 1 - x)) / (inner - outer), 0, 1)
    rows = np.arange(H)[:, None]
    vert = np.clip((rows - y0) / (40 * k), 0, 1) * np.clip((y2 - rows) / (y2 - y1), 0, 1)
    m = (side * vert)[..., None]
    a = a * (1 - m) + ref[:, None, :] * m
    b0, b1, s0, s1 = int(1066 * k), int(1152 * k), int(1118 * k), int(1150 * k)
    sky, stage = a[int(1062 * k)].copy(), a[int(1168 * k)].copy()
    clean = a.copy()
    for y in range(b0, b1):
        t = min(max((y - s0) / (s1 - s0), 0.0), 1.0); t = t * t * (3 - 2 * t)
        clean[y] = sky * (1 - t) + stage * t
    w = np.ones((H, 1, 1), np.float32)
    w[b0:b1] = keep
    fade = int(10 * k)
    for i in range(fade):
        w[b0 + i] = 1 - (1 - keep) * (i + 1) / fade
        w[b1 - 1 - i] = 1 - (1 - keep) * (i + 1) / fade
    band = np.zeros((H, 1, 1), bool); band[b0:b1] = True
    a = np.where(band, clean + (a - clean) * w, a)
    return Image.fromarray(np.clip(a, 0, 255).astype(np.uint8), 'RGB')
